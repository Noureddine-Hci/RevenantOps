spawn_positions = [
    (-2000, -2000, 100),  # Coin SW (zone ouverte)
    (-2000,  2000, 100),  # Coin NW (pres couloir gauche)
    ( 2000, -2000, 100),  # Coin SE

    (    0, -1500, 100),  # Centre-sud (zone ouverte)
]

for i, (x, y, z) in enumerate(spawn_positions):
    actor = unreal.EditorLevelLibrary.spawn_actor_from_class(
        unreal.TargetPoint,
        unreal.Vector(x, y, z),
        unreal.Rotator(0, 0, 0)
    )
    actor.set_actor_label(f'SpawnPoint_{i+1}')
    print(f'Spawned SpawnPoint_{i+1} at ({x}, {y}, {z})')

Les proprietes suivantes doivent etre configurees dans le Details Panel du EnemyWaveSpawner (manuellement si MCP set_property ne fonctionne pas pour les arrays complexes) :

- `bTriggerOnOverlap` = false (demarrage manuel via StartEncounter)

```python
import unreal

# Trouver le spawner
actors = unreal.EditorLevelLibrary.get_all_level_actors()
spawner = None
for a in actors:
    if a.get_actor_label() == 'MercenairesWaveSpawner':
        spawner = a
        break

if spawner:
    spawner.set_editor_property('bTriggerOnOverlap', False)
    spawner.set_editor_property('bInfiniteMode', True)
    spawner.set_editor_property('MaxAliveEnemies', 15)
    print('Spawner properties set.')

    # Trouver les spawn points et les ajouter
    spawn_points = []
    for a in actors:
        label = a.get_actor_label()
        if label and label.startswith('SpawnPoint_'):
            spawn_points.append(a)

    spawn_points.sort(key=lambda a: a.get_actor_label())
    spawner.set_editor_property('SpawnPoints', spawn_points)
    print(f'SpawnPoints assigned: {len(spawn_points)} points')
else:
    print('ERROR: MercenairesWaveSpawner not found')
```
